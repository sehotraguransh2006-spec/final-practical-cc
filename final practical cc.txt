arr = [1, 2, 3, 4]
n = len(arr)
result = [1] * n
prefix = 1
for i in range(n):
    result[i] = prefix
    prefix = prefix * arr[i]
suffix = 1
for i in range(n - 1, -1, -1):
    result[i] = result[i] * suffix
    suffix = suffix * arr[i]
print("Original Array:", arr)
print("Product Except Self:", result)
